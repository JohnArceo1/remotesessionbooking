function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // If this GET request is actually a "save booking" call, handle it here. if (e.parameter && e.parameter.action === 'add' && e.parameter.data) { var booking = JSON.parse(e.parameter.data); sheet.appendRow([ booking.id, booking.name, booking.company, booking.contactMethod, booking.contactValue, booking.date, booking.time, booking.submittedAt ]); return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } // Otherwise, return all bookings. var data = sheet.getDataRange().getValues(); var headers = data[0]; var rows = data.slice(1); var bookings = rows.map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = row[i]; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(bookings)) .setMimeType(ContentService.MimeType.JSON); }
